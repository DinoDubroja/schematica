% Dependent current source; cI.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchSourceDependentCurrent}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{7*\F}
    \SCdimension{body-height}{8*\F}
    \SCdimension{grid}{1}
    \SCpin{a}{0}{ceil(7*\F)}
    \begin{SCpart}{lead-a}
    \draw ({0},{ceil(7*\F)})--({0*\F},{4*\F});
    \end{SCpart}
    \SCpin{b}{0}{floor(-7*\F)}
    \begin{SCpart}{lead-b}
    \draw ({0},{floor(-7*\F)})--({0*\F},{-4*\F});
    \end{SCpart}
    \begin{SCpart}{body}
    \draw ({0*\F},{4*\F})--({3.5*\F},{0*\F})--({0*\F},{-4*\F})--({-3.5*\F},{0*\F})--({0*\F},{4*\F});
    \end{SCpart}
    \begin{SCpart}{current-direction}
    \draw ({0*\F},{-1.9*\F})--({0*\F},{1.9*\F});\begin{scope}[shift={({0*\F},{1.9*\F})},rotate=90]
    \fill ({-1.3*\F},{0.65*\F})--({0*\F},{0*\F})--({-1.3*\F},{-0.65*\F})--cycle;
    \end{scope}
    \end{SCpart}
  \end{scope}
}
