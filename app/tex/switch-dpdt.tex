% DPDT switch; cspst or cspdt; custom ganged contacts.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchSwitchDpdt}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{6*\F}
    \SCdimension{body-height}{14}
    \SCdimension{grid}{1}
    \SCpin{pole-1-common}{floor(-7*\F)}{5}
    \begin{SCpart}{lead-pole-1-common}
    \draw ({floor(-7*\F)},{5})--({-3*\F},{5});
    \end{SCpart}
    \begin{SCpart}{pole-1-pivot}
    \fill ({-3*\F},{5}) circle ({0.4*\F});
    \end{SCpart}
    \SCpin{pole-1-upper}{ceil(7*\F)}{7}
    \begin{SCpart}{lead-pole-1-upper}
    \draw ({ceil(7*\F)},{7})--({3*\F},{7});
    \end{SCpart}
    \begin{SCpart}{pole-1-upper-contact}
    \fill ({3*\F},{7}) circle ({0.4*\F});
    \end{SCpart}
    \SCpin{pole-1-lower}{ceil(7*\F)}{3}
    \begin{SCpart}{lead-pole-1-lower}
    \draw ({ceil(7*\F)},{3})--({3*\F},{3});
    \end{SCpart}
    \begin{SCpart}{pole-1-lower-contact}
    \fill ({3*\F},{3}) circle ({0.4*\F});
    \end{SCpart}
    \begin{SCpart}{pole-1-blade}
    \draw ({-3*\F},{5})--({3*\F},{7});
    \end{SCpart}
    \SCpin{pole-2-common}{floor(-7*\F)}{-5}
    \begin{SCpart}{lead-pole-2-common}
    \draw ({floor(-7*\F)},{-5})--({-3*\F},{-5});
    \end{SCpart}
    \begin{SCpart}{pole-2-pivot}
    \fill ({-3*\F},{-5}) circle ({0.4*\F});
    \end{SCpart}
    \SCpin{pole-2-upper}{ceil(7*\F)}{-3}
    \begin{SCpart}{lead-pole-2-upper}
    \draw ({ceil(7*\F)},{-3})--({3*\F},{-3});
    \end{SCpart}
    \begin{SCpart}{pole-2-upper-contact}
    \fill ({3*\F},{-3}) circle ({0.4*\F});
    \end{SCpart}
    \SCpin{pole-2-lower}{ceil(7*\F)}{-7}
    \begin{SCpart}{lead-pole-2-lower}
    \draw ({ceil(7*\F)},{-7})--({3*\F},{-7});
    \end{SCpart}
    \begin{SCpart}{pole-2-lower-contact}
    \fill ({3*\F},{-7}) circle ({0.4*\F});
    \end{SCpart}
    \begin{SCpart}{pole-2-blade}
    \draw ({-3*\F},{-5})--({3*\F},{-3});
    \end{SCpart}
    \begin{SCpart}{mechanical-link}
    \draw[dash pattern=on 1mm off .7mm] ({0},{-5+1})--({0},{5+1});
    \end{SCpart}
  \end{scope}
}
