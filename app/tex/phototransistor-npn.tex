% NPN phototransistor; npn, photo.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchPhototransistorNpn}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-height}{7*\F}
    \SCdimension{body-width}{4.5*\F}
    \SCdimension{enclosure-diameter}{11.2*\F}
    \SCdimension{grid}{1}
    \SCpin{collector}{ceil(3*\F)}{ceil(6*\F)}
    \begin{SCpart}{lead-collector}
    \draw ({ceil(3*\F)},{ceil(6*\F)})--({ceil(3*\F)},{3.5*\F})--({3*\F},{3.5*\F});
    \end{SCpart}
    \SCpin{emitter}{ceil(3*\F)}{floor(-6*\F)}
    \begin{SCpart}{lead-emitter}
    \draw ({ceil(3*\F)},{floor(-6*\F)})--({ceil(3*\F)},{-3.5*\F})--({3*\F},{-3.5*\F});
    \end{SCpart}
    \begin{SCpart}{base}
    \draw ({-1.5*\F},{-2.4*\F})--({-1.5*\F},{2.4*\F});
    \end{SCpart}
    \begin{SCpart}{collector-branch}
    \draw ({-1.5*\F},{1.5*\F})--({3*\F},{3.5*\F});
    \end{SCpart}
    \begin{SCpart}{emitter-branch}
    \draw ({-1.5*\F},{-1.5*\F})--({3*\F},{-3.5*\F});
    \end{SCpart}
    \begin{SCpart}{emitter-arrow}
    \begin{scope}[shift={({2.2*\F},{-3.14444*\F})},rotate=-23.9625]
    \fill ({-1.05*\F},{0.55*\F})--({0*\F},{0*\F})--({-1.05*\F},{-0.55*\F})--cycle;
    \end{scope}
    \end{SCpart}
    \begin{SCpart}{enclosure}
    \draw ({0.5*\F},{0*\F}) circle ({5.6*\F});
    \end{SCpart}
    \begin{SCpart}{light-arrows}
    \draw ({-8.65702*\F},{5.15702*\F})--({-6.3*\F},{2.8*\F});
    \begin{scope}[shift={({-6.3*\F},{2.8*\F})},rotate=-45]
    \draw ({-1.25*\F},{0.75*\F})--({0*\F},{0*\F})--({-1.25*\F},{-0.75*\F});
    \end{scope}
    \draw ({-6.88926*\F},{6.92479*\F})--({-4.53223*\F},{4.56777*\F});
    \begin{scope}[shift={({-4.53223*\F},{4.56777*\F})},rotate=-45]
    \draw ({-1.25*\F},{0.75*\F})--({0*\F},{0*\F})--({-1.25*\F},{-0.75*\F});
    \end{scope}
    \end{SCpart}
  \end{scope}
}
