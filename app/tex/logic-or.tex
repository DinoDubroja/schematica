% OR gate; or port.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchLogicOr}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{9.3*\F}
    \SCdimension{body-height}{8*\F}
    \SCdimension{grid}{1}
    \SCpin{a}{floor(-8*\F)}{2}
    \begin{SCpart}{lead-a}
    \draw ({floor(-8*\F)},{2})--({-2.5*\F-.5/\F},{2});
    \end{SCpart}
    \SCpin{b}{floor(-8*\F)}{-2}
    \begin{SCpart}{lead-b}
    \draw ({floor(-8*\F)},{-2})--({-2.5*\F-.5/\F},{-2});
    \end{SCpart}
    \SCpin{out}{ceil(9*\F)}{0}
    \begin{SCpart}{lead-out}
    \draw ({ceil(9*\F)},{0})--({4.8*\F},{0*\F});
    \end{SCpart}
    \begin{SCpart}{body}
    \draw ({-4.5*\F},{-4*\F}) .. controls ({-1.83333*\F},{-1.33333*\F}) and ({-1.83333*\F},{1.33333*\F}) .. ({-4.5*\F},{4*\F}) .. controls ({0*\F},{4*\F}) and ({2.4*\F},{3*\F}) .. ({4.8*\F},{0*\F}) .. controls ({2.4*\F},{-3*\F}) and ({0*\F},{-4*\F}) .. ({-4.5*\F},{-4*\F});
    \end{SCpart}
  \end{scope}
}
