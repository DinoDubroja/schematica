% Photodiode; pDo.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchDiodePhoto}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{3.5*\F}
    \SCdimension{body-height}{5*\F}
    \SCdimension{grid}{1}
    \SCpin{a}{floor(-6*\F)}{0}
    \begin{SCpart}{lead-a}
    \draw ({floor(-6*\F)},{0})--({-2*\F},{0*\F});
    \end{SCpart}
    \SCpin{k}{ceil(6*\F)}{0}
    \begin{SCpart}{lead-k}
    \draw ({ceil(6*\F)},{0})--({1.5*\F},{0*\F});
    \end{SCpart}
    \begin{SCpart}{junction}
    \draw ({-2*\F},{-2.5*\F})--({1.5*\F},{0*\F})--({-2*\F},{2.5*\F})--({-2*\F},{-2.5*\F});
    \end{SCpart}
    \begin{SCpart}{cathode}
    \draw ({1.5*\F},{-2.5*\F})--({1.5*\F},{2.5*\F});
    \end{SCpart}
    \begin{SCpart}{light-arrows}
    \draw ({1.35702*\F},{7.20702*\F})--({-1*\F},{4.85*\F});
    \begin{scope}[shift={({-1*\F},{4.85*\F})},rotate=-135]
    \draw ({-1.25*\F},{0.75*\F})--({0*\F},{0*\F})--({-1.25*\F},{-0.75*\F});
    \end{scope}
    \draw ({3.12479*\F},{5.43926*\F})--({0.767767*\F},{3.08223*\F});
    \begin{scope}[shift={({0.767767*\F},{3.08223*\F})},rotate=-135]
    \draw ({-1.25*\F},{0.75*\F})--({0*\F},{0*\F})--({-1.25*\F},{-0.75*\F});
    \end{scope}
    \end{SCpart}
  \end{scope}
}
