% Trimmer capacitor; vC preset-mark adaptation.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchCapacitorTrimmer}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{2*\F}
    \SCdimension{body-height}{6.66667*\F}
    \SCdimension{plate-gap}{2*\F}
    \SCdimension{grid}{1}
    \SCpin{a}{-ceil(\F+2)-1}{0}
    \begin{SCpart}{lead-a}
    \draw ({-ceil(\F+2)-1},{0})--({-1*\F},{0*\F});
    \end{SCpart}
    \SCpin{b}{ceil(\F+2)+1}{0}
    \begin{SCpart}{lead-b}
    \draw ({ceil(\F+2)+1},{0})--({1*\F},{0*\F});
    \end{SCpart}
    \begin{SCpart}{plate-a}
    \draw ({-1*\F},{-3.33333*\F})--({-1*\F},{3.33333*\F});
    \end{SCpart}
    \begin{SCpart}{plate-b}
    \draw ({1*\F},{-3.33333*\F})--({1*\F},{3.33333*\F});
    \end{SCpart}
    \begin{SCpart}{adjustment}
    \begin{scope}[xscale=\SCorientx,yscale=\SCorienty]
    \draw ({-3*\F},{-4.4*\F})--({3*\F},{4.4*\F});\draw ({2.25*\F},{4.91*\F})--({3.75*\F},{3.89*\F});
    \end{scope}
    \end{SCpart}
  \end{scope}
}
