% N-channel depletion MOSFET; nigfetd.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchMosfetNDepletion}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-height}{6*\F}
    \SCdimension{body-width}{3.9*\F}
    \SCdimension{grid}{1}
    \SCpin{gate}{floor(-6*\F)}{0}
    \begin{SCpart}{lead-gate}
    \draw ({floor(-6*\F)},{0})--({-0.9*\F},{0*\F});
    \end{SCpart}
    \SCpin{drain}{ceil(3*\F)}{ceil(6*\F)}
    \begin{SCpart}{lead-drain}
    \draw ({ceil(3*\F)},{ceil(6*\F)})--({ceil(3*\F)},{2.6*\F})--({3*\F},{2.6*\F});
    \end{SCpart}
    \SCpin{source}{ceil(3*\F)}{floor(-6*\F)}
    \begin{SCpart}{lead-source}
    \draw ({ceil(3*\F)},{floor(-6*\F)})--({ceil(3*\F)},{-2.6*\F})--({3*\F},{-2.6*\F});
    \end{SCpart}
    \begin{SCpart}{gate}
    \draw ({-0.9*\F},{-3*\F})--({-0.9*\F},{3*\F});
    \end{SCpart}
    \begin{SCpart}{channel}
    \draw ({0.6*\F},{-3*\F})--({0.6*\F},{3*\F});
    \end{SCpart}
    \begin{SCpart}{drain-branch}
    \draw ({0.6*\F},{2.6*\F})--({3*\F},{2.6*\F});
    \end{SCpart}
    \begin{SCpart}{source-branch}
    \draw ({0.6*\F},{-2.6*\F})--({3*\F},{-2.6*\F});
    \end{SCpart}
    \begin{SCpart}{body-source-link}
    \draw ({0.6*\F},{0*\F})--({3*\F},{0*\F})--({3*\F},{-2.6*\F});\fill ({3*\F},{-2.6*\F}) circle ({0.28*\F});
    \end{SCpart}
    \begin{SCpart}{body-arrow}
    \begin{scope}[shift={({0.6*\F},{0*\F})},rotate=180]
    \fill ({-1.05*\F},{0.55*\F})--({0*\F},{0*\F})--({-1.05*\F},{-0.55*\F})--cycle;
    \end{scope}
    \end{SCpart}
  \end{scope}
}
