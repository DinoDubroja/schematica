% Schmitt-trigger buffer; schmitt.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchLogicSchmittBuffer}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{9*\F}
    \SCdimension{body-height}{8*\F}
    \SCdimension{grid}{1}
    \SCpin{in}{floor(-8*\F)}{0}
    \begin{SCpart}{lead-in}
    \draw ({floor(-8*\F)},{0})--({-4.5*\F},{0*\F});
    \end{SCpart}
    \SCpin{out}{ceil(9*\F)}{0}
    \begin{SCpart}{lead-out}
    \draw ({ceil(9*\F)},{0})--({4.5*\F},{0*\F});
    \end{SCpart}
    \begin{SCpart}{body}
    \draw ({-4.5*\F},{-4*\F})--({-4.5*\F},{4*\F})--({4.5*\F},{0*\F})--({-4.5*\F},{-4*\F});
    \end{SCpart}
    \begin{SCpart}{hysteresis}
    \draw ({-2*\F},{-1.2*\F}) rectangle ({-0.8*\F},{1.2*\F});
    \draw ({-3*\F},{-1.2*\F})--({-2*\F},{-1.2*\F});
    \draw ({-0.8*\F},{1.2*\F})--({-0.2*\F},{1.2*\F});
    \end{SCpart}
  \end{scope}
}
