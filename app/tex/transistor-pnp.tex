% PNP transistor; pnp.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchTransistorPnp}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  % A fixed 1 mm left shift at every size; recompute the emitter direction.
  \pgfmathsetmacro{\SCbranchangle}{atan2(-2*\F,4.5*\F-1)}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-height}{7*\F}
    \SCdimension{body-width}{4.5*\F-1}
    \SCdimension{grid}{1}
    \SCpin{base}{floor(-6*\F)}{0}
    \begin{SCpart}{lead-base}
    \draw ({floor(-6*\F)},{0})--({-1.5*\F},{0*\F});
    \end{SCpart}
    \SCpin{collector}{ceil(3*\F)-1}{ceil(6*\F)}
    \begin{SCpart}{lead-collector}
    \draw ({ceil(3*\F)-1},{ceil(6*\F)})--({ceil(3*\F)-1},{3.5*\F})--({3*\F-1},{3.5*\F});
    \end{SCpart}
    \SCpin{emitter}{ceil(3*\F)-1}{floor(-6*\F)}
    \begin{SCpart}{lead-emitter}
    \draw ({ceil(3*\F)-1},{floor(-6*\F)})--({ceil(3*\F)-1},{-3.5*\F})--({3*\F-1},{-3.5*\F});
    \end{SCpart}
    \begin{SCpart}{base}
    \draw ({-1.5*\F},{-2.4*\F})--({-1.5*\F},{2.4*\F});
    \end{SCpart}
    \begin{SCpart}{collector-branch}
    \draw ({-1.5*\F},{1.5*\F})--({3*\F-1},{3.5*\F});
    \end{SCpart}
    \begin{SCpart}{emitter-branch}
    \draw ({-1.5*\F},{-1.5*\F})--({3*\F-1},{-3.5*\F});
    \end{SCpart}
    \begin{SCpart}{emitter-arrow}
    \begin{scope}[shift={({3*\F-1-1.05*\F*cos(\SCbranchangle)},{-3.5*\F-1.05*\F*sin(\SCbranchangle)})},rotate={\SCbranchangle+180}]
    \fill ({-1.05*\F},{0.55*\F})--({0*\F},{0*\F})--({-1.05*\F},{-0.55*\F})--cycle;
    \end{scope}
    \end{SCpart}
  \end{scope}
}
