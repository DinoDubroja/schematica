% DPST switch; cspst or cspdt; custom ganged contacts.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchSwitchDpst}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{6*\F}
    \SCdimension{body-height}{10.2}
    \SCdimension{grid}{1}
    \SCpin{pole-1-common}{floor(-7*\F)}{4}
    \begin{SCpart}{lead-pole-1-common}
    \draw ({floor(-7*\F)},{4})--({-3*\F},{4});
    \end{SCpart}
    \begin{SCpart}{pole-1-pivot}
    \fill ({-3*\F},{4}) circle ({0.4*\F});
    \end{SCpart}
    \SCpin{pole-1-out}{ceil(7*\F)}{4}
    \begin{SCpart}{lead-pole-1-out}
    \draw ({ceil(7*\F)},{4})--({3*\F},{4});
    \end{SCpart}
    \begin{SCpart}{pole-1-fixed-contact}
    \fill ({3*\F},{4}) circle ({0.4*\F});
    \end{SCpart}
    \begin{SCpart}{pole-1-blade}
    \draw ({-3*\F},{4})--({2.2*\F},{4+2.2*\F});
    \end{SCpart}
    \SCpin{pole-2-common}{floor(-7*\F)}{-4}
    \begin{SCpart}{lead-pole-2-common}
    \draw ({floor(-7*\F)},{-4})--({-3*\F},{-4});
    \end{SCpart}
    \begin{SCpart}{pole-2-pivot}
    \fill ({-3*\F},{-4}) circle ({0.4*\F});
    \end{SCpart}
    \SCpin{pole-2-out}{ceil(7*\F)}{-4}
    \begin{SCpart}{lead-pole-2-out}
    \draw ({ceil(7*\F)},{-4})--({3*\F},{-4});
    \end{SCpart}
    \begin{SCpart}{pole-2-fixed-contact}
    \fill ({3*\F},{-4}) circle ({0.4*\F});
    \end{SCpart}
    \begin{SCpart}{pole-2-blade}
    \draw ({-3*\F},{-4})--({2.2*\F},{-4+2.2*\F});
    \end{SCpart}
    \begin{SCpart}{mechanical-link}
    \draw[dash pattern=on 1mm off .7mm] ({0},{-4+1.26923*\F})--({0},{4+1.26923*\F});
    \end{SCpart}
  \end{scope}
}
