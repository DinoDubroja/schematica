% Relay coil; custom relay coil/contact assembly.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchRelayCoil}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{5*\F}
    \SCdimension{body-height}{6*\F}
    \SCdimension{grid}{1}
    \begin{SCpart}{coil-body}
    \draw ({-2.5*\F},{-3*\F}) rectangle ({2.5*\F},{3*\F});
    \end{SCpart}
    \begin{SCpart}{coil-label}
    \pgfmathsetmacro{\SCfontsize}{1.8*\F*72.27/25.4/.69444}
    \font\SCbatchfont="[lmsans10-regular.otf]" at \SCfontsize pt
    \node[inner sep=0pt,outer sep=0pt] at ({0*\F},{0*\F}) {\SCbatchfont K};
    \end{SCpart}
    \SCpin{coil-1}{0}{ceil(6*\F)}
    \begin{SCpart}{lead-coil-1}
    \draw ({0},{ceil(6*\F)})--({0},{3*\F})--({0*\F},{3*\F});
    \end{SCpart}
    \SCpin{coil-2}{0}{floor(-6*\F)}
    \begin{SCpart}{lead-coil-2}
    \draw ({0},{floor(-6*\F)})--({0},{-3*\F})--({0*\F},{-3*\F});
    \end{SCpart}
  \end{scope}
}
