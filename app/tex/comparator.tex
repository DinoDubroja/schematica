% Comparator; custom labelled comparator functional block.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchComparator}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{10*\F}
    \SCdimension{body-height}{12*\F}
    \SCdimension{grid}{1}
    \begin{SCpart}{body}
    \draw ({-5*\F},{-6*\F}) rectangle ({5*\F},{6*\F});
    \end{SCpart}
    \SCpin{in-plus}{floor(-8*\F)}{3}
    \begin{SCpart}{lead-in-plus}
    \draw ({floor(-8*\F)},{3})--({-5*\F},{3});
    \end{SCpart}
    \begin{SCpart}{input-plus}
    \begin{scope}[yshift=3mm]
    \draw ({-4.05*\F},{0*\F})--({-3.15*\F},{0*\F});
    \draw ({-3.6*\F},{-0.45*\F})--({-3.6*\F},{0.45*\F});
    \end{scope}
    \end{SCpart}
    \SCpin{in-minus}{floor(-8*\F)}{-3}
    \begin{SCpart}{lead-in-minus}
    \draw ({floor(-8*\F)},{-3})--({-5*\F},{-3});
    \end{SCpart}
    \begin{SCpart}{input-minus}
    \begin{scope}[yshift=-3mm]
    \draw ({-4.05*\F},{0*\F})--({-3.15*\F},{0*\F});
    \end{scope}
    \end{SCpart}
    \SCpin{out}{ceil(8*\F)}{0}
    \begin{SCpart}{lead-out}
    \draw ({ceil(8*\F)},{0})--({5*\F},{0*\F});
    \end{SCpart}
    \begin{SCpart}{function}
    \pgfmathsetmacro{\SCfontsize}{1.6*\F*72.27/25.4/.69444}
    \font\SCbatchfont="[lmsans10-regular.otf]" at \SCfontsize pt
    \node[inner sep=0pt,outer sep=0pt] at ({0.4*\F},{0*\F}) {\SCbatchfont CMP};
    \end{SCpart}
  \end{scope}
}
