% Shared accepted palette kinds 4 and 5; final argument enables supply leads.
\newcommand{\SCOpampBase}[4]{%
  \SCresolve{#1}{#2}{#3}%
  \pgfmathsetmacro{\SChalfwidth}{3.4*\SCscale}%
  \pgfmathsetmacro{\SCend}{ceil(\SChalfwidth+2)+1}%
  \pgfmathsetmacro{\SCsupplyend}{ceil(2*\SCscale+2)}%
  \pgfmathsetmacro{\SCmarkx}{-\SChalfwidth+\SCscale}%
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCpin{in-plus}{-\SCend}{3}\SCpin{in-minus}{-\SCend}{-3}
    \SCpin{out}{\SCend}{0}
    \ifnum#4=1 \SCpin{supply-plus}{0}{\SCsupplyend}\SCpin{supply-minus}{0}{-\SCsupplyend}\fi
    \SCdimension{body-width}{2*\SChalfwidth}
    \SCdimension{body-height}{8*\SCscale}
    \begin{SCpart}{body}
      \draw (-\SChalfwidth,{-4*\SCscale})--(-\SChalfwidth,{4*\SCscale})--(\SChalfwidth,0)--cycle;
    \end{SCpart}
    \begin{SCpart}{lead-in-plus}\draw (-\SCend,3)--(-\SChalfwidth,3);\end{SCpart}
    \begin{SCpart}{lead-in-minus}\draw (-\SCend,-3)--(-\SChalfwidth,-3);\end{SCpart}
    \begin{SCpart}{lead-out}\draw (\SChalfwidth,0)--(\SCend,0);\end{SCpart}
    \begin{SCpart}{sign-plus}
      \draw ({\SCmarkx-0.4*\SCscale},3)--({\SCmarkx+0.4*\SCscale},3)
        (\SCmarkx,{3-0.4*\SCscale})--(\SCmarkx,{3+0.4*\SCscale});
    \end{SCpart}
    \begin{SCpart}{sign-minus}
      \draw ({\SCmarkx-0.4*\SCscale},-3)--({\SCmarkx+0.4*\SCscale},-3);
    \end{SCpart}
    \ifnum#4=1
    \begin{SCpart}{lead-supply-plus}\draw (0,{2*\SCscale})--(0,\SCsupplyend);\end{SCpart}
    \begin{SCpart}{lead-supply-minus}\draw (0,{-2*\SCscale})--(0,-\SCsupplyend);\end{SCpart}
    \fi
  \end{scope}%
}
