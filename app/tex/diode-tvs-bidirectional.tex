% Bidirectional TVS; custom back-to-back avalanche pair.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchDiodeTvsBidirectional}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{7.6*\F}
    \SCdimension{body-height}{5*\F}
    \SCdimension{grid}{1}
    \SCpin{a}{floor(-7*\F)}{0}
    \begin{SCpart}{lead-a}
    \draw ({floor(-7*\F)},{0})--({-3.8*\F},{0*\F});
    \end{SCpart}
    \SCpin{b}{ceil(7*\F)}{0}
    \begin{SCpart}{lead-b}
    \draw ({ceil(7*\F)},{0})--({3.8*\F},{0*\F});
    \end{SCpart}
    \begin{SCpart}{left-junction}
    \draw ({-3.8*\F},{-2.5*\F})--({-1.2*\F},{0*\F})--({-3.8*\F},{2.5*\F})--({-3.8*\F},{-2.5*\F});
    \end{SCpart}
    \begin{SCpart}{left-cathode}
    \draw ({-0.55*\F},{-3*\F})--({-1.2*\F},{-2.5*\F})--({-1.2*\F},{2.5*\F})--({-1.85*\F},{3*\F});
    \end{SCpart}
    \begin{SCpart}{right-junction}
    \draw ({3.8*\F},{-2.5*\F})--({1.2*\F},{0*\F})--({3.8*\F},{2.5*\F})--({3.8*\F},{-2.5*\F});
    \end{SCpart}
    \begin{SCpart}{right-cathode}
    \draw ({0.55*\F},{-3*\F})--({1.2*\F},{-2.5*\F})--({1.2*\F},{2.5*\F})--({1.85*\F},{3*\F});
    \end{SCpart}
    \begin{SCpart}{junction-link}
    \draw ({-1.2*\F},{0*\F})--({1.2*\F},{0*\F});
    \end{SCpart}
  \end{scope}
}
