% Coaxial connector; coaxport concept; separate shield and center.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchConnectorCoaxial}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{4.4*\F}
    \SCdimension{body-height}{4.4*\F}
    \SCdimension{grid}{1}
    \SCpin{center}{floor(-6*\F)}{0}
    \begin{SCpart}{lead-center}
    \draw ({floor(-6*\F)},{0})--({-0.65*\F},{0*\F});
    \end{SCpart}
    \SCpin{shield}{0}{floor(-5*\F)}
    \begin{SCpart}{lead-shield}
    \draw ({0},{floor(-5*\F)})--({0*\F},{-2.2*\F});
    \end{SCpart}
    \begin{SCpart}{center-contact}
    \draw ({0*\F},{0*\F}) circle ({0.65*\F});
    \end{SCpart}
    \begin{SCpart}{shield-contact}
    \draw ({-1.90526*\F},{1.1*\F}) arc[start angle=150,end angle=-150,radius={2.2*\F}];
    \end{SCpart}
  \end{scope}
}
