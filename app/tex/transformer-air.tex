% Air-core transformer; transformer.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchTransformerAir}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{6.4*\F}
    \SCdimension{body-height}{10*\F}
    \SCdimension{grid}{1}
    \SCpin{primary-1}{floor(-7*\F)}{ceil(5*\F)}
    \begin{SCpart}{lead-primary-1}
    \draw ({floor(-7*\F)},{ceil(5*\F)})--({floor(-7*\F)},{5*\F})--({-3.2*\F},{5*\F});
    \end{SCpart}
    \SCpin{primary-2}{floor(-7*\F)}{floor(-5*\F)}
    \begin{SCpart}{lead-primary-2}
    \draw ({floor(-7*\F)},{floor(-5*\F)})--({floor(-7*\F)},{-5*\F})--({-3.2*\F},{-5*\F});
    \end{SCpart}
    \begin{SCpart}{primary-winding}
    \draw ({-3.2*\F},{-5*\F}) .. controls ({-1.2*\F},{-5*\F}) and ({-1.2*\F},{-2.5*\F}) .. ({-3.2*\F},{-2.5*\F}) .. controls ({-1.2*\F},{-2.5*\F}) and ({-1.2*\F},{0*\F}) .. ({-3.2*\F},{0*\F}) .. controls ({-1.2*\F},{0*\F}) and ({-1.2*\F},{2.5*\F}) .. ({-3.2*\F},{2.5*\F}) .. controls ({-1.2*\F},{2.5*\F}) and ({-1.2*\F},{5*\F}) .. ({-3.2*\F},{5*\F});
    \end{SCpart}
    \begin{SCpart}{primary-polarity}
    \fill ({-4.3*\F},{3.8*\F}) circle ({0.28*\F});
    \end{SCpart}
    \SCpin{secondary-1}{ceil(7*\F)}{ceil(5*\F)}
    \begin{SCpart}{lead-secondary-1}
    \draw ({ceil(7*\F)},{ceil(5*\F)})--({ceil(7*\F)},{5*\F})--({3.2*\F},{5*\F});
    \end{SCpart}
    \SCpin{secondary-2}{ceil(7*\F)}{floor(-5*\F)}
    \begin{SCpart}{lead-secondary-2}
    \draw ({ceil(7*\F)},{floor(-5*\F)})--({ceil(7*\F)},{-5*\F})--({3.2*\F},{-5*\F});
    \end{SCpart}
    \begin{SCpart}{secondary-winding}
    \draw ({3.2*\F},{-5*\F}) .. controls ({1.2*\F},{-5*\F}) and ({1.2*\F},{-2.5*\F}) .. ({3.2*\F},{-2.5*\F}) .. controls ({1.2*\F},{-2.5*\F}) and ({1.2*\F},{0*\F}) .. ({3.2*\F},{0*\F}) .. controls ({1.2*\F},{0*\F}) and ({1.2*\F},{2.5*\F}) .. ({3.2*\F},{2.5*\F}) .. controls ({1.2*\F},{2.5*\F}) and ({1.2*\F},{5*\F}) .. ({3.2*\F},{5*\F});
    \end{SCpart}
    \begin{SCpart}{secondary-polarity}
    \fill ({4.3*\F},{3.8*\F}) circle ({0.28*\F});
    \end{SCpart}
  \end{scope}
}
