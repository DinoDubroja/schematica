% Voltage source; V.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchSourceVoltage}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{7.2*\F}
    \SCdimension{body-height}{7.2*\F}
    \SCdimension{grid}{1}
    \SCpin{positive}{0}{ceil(7*\F)}
    \begin{SCpart}{lead-positive}
    \draw ({0},{ceil(7*\F)})--({0*\F},{3.6*\F});
    \end{SCpart}
    \SCpin{negative}{0}{floor(-7*\F)}
    \begin{SCpart}{lead-negative}
    \draw ({0},{floor(-7*\F)})--({0*\F},{-3.6*\F});
    \end{SCpart}
    \begin{SCpart}{body}
    \draw ({0*\F},{0*\F}) circle ({3.6*\F});
    \end{SCpart}
    \begin{SCpart}{positive-mark}
    \draw ({-0.65*\F},{1.55*\F})--({0.65*\F},{1.55*\F});
    \draw ({0*\F},{0.9*\F})--({0*\F},{2.2*\F});
    \end{SCpart}
    \begin{SCpart}{negative-mark}
    \draw ({-0.65*\F},{-1.55*\F})--({0.65*\F},{-1.55*\F});
    \end{SCpart}
  \end{scope}
}
