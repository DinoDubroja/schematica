% Polarized capacitor; eC.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchCapacitorPolarized}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{2*\F}
    \SCdimension{body-height}{6.66667*\F}
    \SCdimension{plate-gap}{2*\F}
    \SCdimension{grid}{1}
    \SCpin{positive}{-ceil(\F+2)-1}{0}
    \begin{SCpart}{lead-positive}
    \draw ({-ceil(\F+2)-1},{0})--({-1*\F},{0*\F});
    \end{SCpart}
    \SCpin{negative}{ceil(\F+2)+1}{0}
    \begin{SCpart}{lead-negative}
    \draw ({ceil(\F+2)+1},{0})--({1*\F},{0*\F});
    \end{SCpart}
    \begin{SCpart}{plate-positive}
    \draw ({-1*\F},{-3.33333*\F})--({-1*\F},{3.33333*\F});
    \end{SCpart}
    \begin{SCpart}{plate-negative}
    \draw ({1*\F},{-3.33333*\F})--({1*\F},{3.33333*\F});
    \end{SCpart}
    \begin{SCpart}{polarity}
    \draw ({-3.2*\F},{2.65*\F})--({-2.1*\F},{2.65*\F});
    \draw ({-2.65*\F},{2.1*\F})--({-2.65*\F},{3.2*\F});
    \end{SCpart}
  \end{scope}
}
