% CircuitikZ reference forms adapted to the accepted Schematica resistor bodies.
% See docs/RESISTOR_BATCH.md for meaning, reference mapping and adaptations.
% All adornments use size coordinates; none depends geometrically on line width.
\newcommand{\SCResistorVariant}[6]{%
  \SCresolve{#1}{#2}{#3}%
  \pgfmathsetmacro{\SCend}{ceil(3*\SCscale+2)+1}%
  \pgfmathsetmacro{\SCwiperend}{ceil(3*\SCscale)+1}%
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#6,2)==1,-1,1)}%
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#6>1,-1,1)}%
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCpin{a}{-\SCend}{0}\SCpin{b}{\SCend}{0}
    \SCdimension{body-width}{6*\SCscale}\SCdimension{body-height}{2*\SCscale}
    \begin{SCpart}{body}
      \ifnum#4=0
        \draw ({-3*\SCscale},-\SCscale) rectangle ({3*\SCscale},\SCscale);
      \else
        \draw ({-3*\SCscale},0)--({-2.5*\SCscale},\SCscale)--({-1.5*\SCscale},-\SCscale)
          --({-0.5*\SCscale},\SCscale)--({0.5*\SCscale},-\SCscale)
          --({1.5*\SCscale},\SCscale)--({2.5*\SCscale},-\SCscale)--({3*\SCscale},0);
      \fi
    \end{SCpart}
    \begin{SCpart}{lead-a}\draw (-\SCend,0)--({-3*\SCscale},0);\end{SCpart}
    \begin{SCpart}{lead-b}\draw ({3*\SCscale},0)--(\SCend,0);\end{SCpart}
    \ifcase#5
      \begin{SCpart}{adjustment}
        \begin{scope}[xscale=\SCorientx,yscale=\SCorienty]
          \draw ({-2*\SCscale},{-2*\SCscale})--({2.4*\SCscale},{2.4*\SCscale});
          \SCopenArrowHead{2.4*\SCscale}{2.4*\SCscale}{45}
        \end{scope}
      \end{SCpart}
    \or \SCresistorWiper
    \or
      \begin{SCpart}{adjustment}
        \begin{scope}[xscale=\SCorientx,yscale=\SCorienty]
          \draw ({-2*\SCscale},{-2*\SCscale})--({2*\SCscale},{2*\SCscale});
          \draw ({1.5*\SCscale},{2.5*\SCscale})--({2.5*\SCscale},{1.5*\SCscale});
        \end{scope}
      \end{SCpart}
    \or
      \SCresistorWiper
      \begin{SCpart}{adjustment}
        \draw ({-.55*\SCscale},{\SCwipertip+1.1*\SCscale})--({.55*\SCscale},{\SCwipertip+1.7*\SCscale});
      \end{SCpart}
    \or \SCresistorDependence
    \or \SCresistorDependence\SCtemperatureMark{NTC}
    \or \SCresistorDependence\SCtemperatureMark{PTC}
    \or
      \begin{SCpart}{nonlinear-mark}
        \begin{scope}[rotate=-45]
          % Selected option B: one cubic, with no internal join.
          % At medium size: axial half-length 5.6 mm, lateral controls 3.4 mm.
          \draw (0,{-3.36*\SCscale})
            .. controls ({2.04*\SCscale},{-1.12*\SCscale})
            and ({-2.04*\SCscale},{1.12*\SCscale}) .. (0,{3.36*\SCscale});
        \end{scope}
      \end{SCpart}
    \or
      \SCphotoArrows{1.8}{2.3}
    \or
      \SCdimension{enclosure-diameter}{7.6*\SCscale}
      \begin{SCpart}{enclosure}\draw (0,0) circle ({3.8*\SCscale});\end{SCpart}
      \SCphotoArrows{3.1}{3.1}
    \fi
  \end{scope}%
}
\newcommand{\SCresistorWiper}{%
  \SCpin{wiper}{0}{\SCwiperend}
  \pgfmathsetmacro{\SCwipertip}{\SCscale+1}%
  \SCdimension{wiper-tip-x}{0}\SCdimension{wiper-tip-y}{\SCwipertip}
  \SCdimension{wiper-body-gap}{1}
  \begin{SCpart}{lead-wiper}\draw (0,\SCwiperend)--(0,\SCwipertip);\end{SCpart}
  \begin{SCpart}{wiper-arrow}
    \SCopenArrowHead{0}{\SCwipertip}{-90}
  \end{SCpart}
}
\newcommand{\SCresistorDependence}{%
  \begin{SCpart}{dependency-mark}
    \draw ({-2.8*\SCscale},{-1.8*\SCscale})--({-1.6*\SCscale},{-1.8*\SCscale})--({2.2*\SCscale},{1.6*\SCscale});
  \end{SCpart}
}
\newcommand{\SCtemperatureMark}[1]{%
  \begin{SCpart}{temperature-mark}
    \pgfmathsetmacro{\SCcapheight}{.84*\SCscale}%
    % Latin Modern Sans capitals have height 0.69444 em. Set the font itself;
    % graphicx resize specials are not portable through the XDV converter.
    \pgfmathsetmacro{\SCfontsize}{\SCcapheight*72.27/25.4/.69444}%
    \font\SCtemperaturefont="[lmsans10-regular.otf]" at \SCfontsize pt
    \node[anchor=north west,inner sep=0pt,outer sep=0pt] at ({-3.2*\SCscale},{-2.4*\SCscale})
      {\SCtemperaturefont #1};
  \end{SCpart}
}
% Axis-based open head: same dimensions for every arrow in this family.
\newcommand{\SCopenArrowHead}[3]{%
  \begin{scope}[shift={(#1,#2)},rotate=#3]
    \draw ({-.75*\SCscale},{.45*\SCscale})--(0,0)--({-.75*\SCscale},{-.45*\SCscale});
  \end{scope}%
}
\newcommand{\SCphotoArrows}[2]{%
  \begin{SCpart}{light-arrows}
    \foreach \offset in {-.75,.75}{%
      \pgfmathsetmacro{\SCax}{(#1+\offset/sqrt(2))*\SCscale}%
      \pgfmathsetmacro{\SCay}{(#2-\offset/sqrt(2))*\SCscale}%
      \draw ({\SCax+sqrt(2)*\SCscale},{\SCay+sqrt(2)*\SCscale})--(\SCax,\SCay);
      \SCopenArrowHead{\SCax}{\SCay}{225}
    }
  \end{SCpart}%
}
