% Voltage regulator; custom three-terminal regulator functional block.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchRegulator}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{10*\F}
    \SCdimension{body-height}{8*\F}
    \SCdimension{grid}{1}
    \begin{SCpart}{body}
    \draw ({-5*\F},{-4*\F}) rectangle ({5*\F},{4*\F});
    \end{SCpart}
    \SCpin{in}{floor(-8*\F)}{0}
    \begin{SCpart}{lead-in}
    \draw ({floor(-8*\F)},{0})--({-5*\F},{0*\F});
    \end{SCpart}
    \SCpin{out}{ceil(8*\F)}{0}
    \begin{SCpart}{lead-out}
    \draw ({ceil(8*\F)},{0})--({5*\F},{0*\F});
    \end{SCpart}
    \SCpin{common}{0}{floor(-7*\F)}
    \begin{SCpart}{lead-common}
    \draw ({0},{floor(-7*\F)})--({0*\F},{-4*\F});
    \end{SCpart}
    \begin{SCpart}{function}
    \pgfmathsetmacro{\SCfontsize}{1.6*\F*72.27/25.4/.69444}
    \font\SCbatchfont="[lmsans10-regular.otf]" at \SCfontsize pt
    \node[inner sep=0pt,outer sep=0pt] at ({0*\F},{2.3*\F}) {\SCbatchfont REG};
    \end{SCpart}
    \begin{SCpart}{input-label}
    \pgfmathsetmacro{\SCfontsize}{1.4*\F*72.27/25.4/.69444}
    \font\SCbatchfont="[lmsans10-regular.otf]" at \SCfontsize pt
    \node[inner sep=0pt,outer sep=0pt] at ({-3*\F},{0*\F}) {\SCbatchfont IN};
    \end{SCpart}
    \begin{SCpart}{output-label}
    \pgfmathsetmacro{\SCfontsize}{1.4*\F*72.27/25.4/.69444}
    \font\SCbatchfont="[lmsans10-regular.otf]" at \SCfontsize pt
    \node[inner sep=0pt,outer sep=0pt] at ({2.2*\F},{0*\F}) {\SCbatchfont OUT};
    \end{SCpart}
    \begin{SCpart}{common-label}
    \pgfmathsetmacro{\SCfontsize}{1.4*\F*72.27/25.4/.69444}
    \font\SCbatchfont="[lmsans10-regular.otf]" at \SCfontsize pt
    \node[inner sep=0pt,outer sep=0pt] at ({0*\F},{-2.6*\F}) {\SCbatchfont COM};
    \end{SCpart}
  \end{scope}
}
