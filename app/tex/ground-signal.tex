% Signal ground; sground.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchGroundSignal}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{5.4*\F}
    \SCdimension{body-height}{2.4*\F}
    \SCdimension{grid}{1}
    \SCpin{reference}{0}{ceil(3*\F)}
    \begin{SCpart}{lead-reference}
    \draw ({0},{ceil(3*\F)})--({0*\F},{0*\F});
    \end{SCpart}
    \begin{SCpart}{ground-mark}
    \draw ({-2.7*\F},{0*\F})--({2.7*\F},{0*\F})--({0*\F},{-2.4*\F})--({-2.7*\F},{0*\F});
    \end{SCpart}
  \end{scope}
}
