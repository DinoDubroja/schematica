% Test point; custom labelled test-point marker.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchTestPoint}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-width}{2*\F}
    \SCdimension{body-height}{4.2*\F}
    \SCdimension{grid}{1}
    \SCpin{net}{0}{floor(-5*\F)}
    \begin{SCpart}{lead-net}
    \draw ({0},{floor(-5*\F)})--({0*\F},{-1*\F});
    \end{SCpart}
    \begin{SCpart}{probe-contact}
    \draw ({0*\F},{0*\F}) circle ({1*\F});
    \end{SCpart}
    \begin{SCpart}{test-label}
    \pgfmathsetmacro{\SCfontsize}{1.4*\F*72.27/25.4/.69444}
    \font\SCbatchfont="[lmsans10-regular.otf]" at \SCfontsize pt
    \node[inner sep=0pt,outer sep=0pt] at ({0*\F},{2.5*\F}) {\SCbatchfont TP};
    \end{SCpart}
  \end{scope}
}
