% Feedthrough capacitor; custom three-terminal feedthrough form.
% Schematica coordinates are millimetres at Medium; pins snap outward to the 1 mm grid.
\newcommand{\SCBatchCapacitorFeedthrough}[4]{%
  \SCresolve{#1}{#2}{#3}
  \def\F{#2}
  \pgfmathsetmacro{\SCorientx}{ifthenelse(mod(#4,2)==1,-1,1)}
  \pgfmathsetmacro{\SCorienty}{ifthenelse(#4>1,-1,1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{plate-gap}{1.3*\F}
    \SCdimension{body-width}{6*\F}
    \SCdimension{body-height}{4.2*\F}
    \SCdimension{grid}{1}
    \SCpin{in}{floor(-5*\F)}{0}
    \begin{SCpart}{lead-in}
    \draw ({floor(-5*\F)},{0})--({-2*\F},{0*\F});
    \end{SCpart}
    \SCpin{out}{ceil(5*\F)}{0}
    \begin{SCpart}{lead-out}
    \draw ({ceil(5*\F)},{0})--({2*\F},{0*\F});
    \end{SCpart}
    \SCpin{case}{0}{floor(-5*\F)}
    \begin{SCpart}{lead-case}
    \draw ({0},{floor(-5*\F)})--({0*\F},{-1.3*\F});
    \end{SCpart}
    \begin{SCpart}{through-conductor}
    \draw ({-2*\F},{0*\F})--({2*\F},{0*\F});
    \end{SCpart}
    \begin{SCpart}{case-plate}
    \draw ({-2*\F},{-1.3*\F})--({2*\F},{-1.3*\F});
    \end{SCpart}
    \begin{SCpart}{enclosure}
    \draw ({-3*\F},{-2.5*\F}) rectangle ({3*\F},{1.7*\F});
    \end{SCpart}
  \end{scope}
}
