\newcommand{\SCCustom}[3]{%
\SCresolve{#1}{#2}{#3}
\pgfmathsetmacro{\SCF}{#2}

\begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
\SCdimension{grid}{1}
\SCpin{gate}{floor(-6*\SCF)}{0}
\SCpin{drain}{ceil(3*\SCF)}{ceil(6*\SCF)}
\SCpin{source}{ceil(3*\SCF)}{floor(-6*\SCF)}
\begin{SCpart}{enclosure}
\draw ({0.5*\SCF},{0*\SCF}) circle ({5.6*\SCF});
\end{SCpart}
\begin{SCpart}{channel}
\draw[line width={1.5*\SCstroke mm}] ({-0.6*\SCF},{-3*\SCF})--({-0.6*\SCF},{3*\SCF});
\end{SCpart}
\begin{SCpart}{lead-gate}
\draw ({floor(-6*\SCF)},{0})--({-0.6*\SCF},{0})--({-0.6*\SCF},{0*\SCF});
\end{SCpart}
\begin{SCpart}{lead-drain}
\draw ({ceil(3*\SCF)},{ceil(6*\SCF)})--({ceil(3*\SCF)},{2.6*\SCF})--({3*\SCF},{2.6*\SCF});
\end{SCpart}
\begin{SCpart}{lead-source}
\draw ({ceil(3*\SCF)},{floor(-6*\SCF)})--({ceil(3*\SCF)},{-2.6*\SCF})--({3*\SCF},{-2.6*\SCF});
\end{SCpart}
\begin{SCpart}{drain-branch}
\draw ({-0.6*\SCF},{2.6*\SCF})--({3*\SCF},{2.6*\SCF});
\end{SCpart}
\begin{SCpart}{source-branch}
\draw ({-0.6*\SCF},{-2.6*\SCF})--({3*\SCF},{-2.6*\SCF});
\end{SCpart}
\begin{SCpart}{gate-arrow}
\fill ({-1.05*\SCF},{-0.5500000000000002*\SCF})--({-2.1*\SCF},{0*\SCF})--({-1.05*\SCF},{0.5499999999999999*\SCF})--cycle;
\end{SCpart}
\end{scope}
}
