\newcommand{\SCCustom}[3]{%
  \SCresolve{#1}{#2}{#3}
  \pgfmathsetmacro{\SCfactor}{#2}
  \pgfmathsetmacro{\SCbranchangle}{atan2(-2*\SCfactor,4.5*\SCfactor-1)}
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCdimension{body-height}{7*\SCfactor}
    \SCdimension{body-width}{4.5*\SCfactor-1}
    \SCdimension{grid}{1}
    \SCpin{base}{floor(-6*\SCfactor)}{0}
    \begin{SCpart}{lead-base}
      \draw ({floor(-6*\SCfactor)},0)--({-1.5*\SCfactor},{0*\SCfactor});
    \end{SCpart}
    \SCpin{collector}{ceil(3*\SCfactor)-1}{ceil(6*\SCfactor)}
    \begin{SCpart}{lead-collector}
      \draw ({ceil(3*\SCfactor)-1},{ceil(6*\SCfactor)})--({ceil(3*\SCfactor)-1},{3.5*\SCfactor})--({3*\SCfactor-1},{3.5*\SCfactor});
    \end{SCpart}
    \SCpin{emitter}{ceil(3*\SCfactor)-1}{floor(-6*\SCfactor)}
    \begin{SCpart}{lead-emitter}
      \draw ({ceil(3*\SCfactor)-1},{floor(-6*\SCfactor)})--({ceil(3*\SCfactor)-1},{-3.5*\SCfactor})--({3*\SCfactor-1},{-3.5*\SCfactor});
    \end{SCpart}
    \begin{SCpart}{base}
      \draw ({-1.5*\SCfactor},{-2.4*\SCfactor})--({-1.5*\SCfactor},{2.4*\SCfactor});
    \end{SCpart}
    \begin{SCpart}{collector-branch}
      \draw ({-1.5*\SCfactor},{1.5*\SCfactor})--({3*\SCfactor-1},{3.5*\SCfactor});
    \end{SCpart}
    \begin{SCpart}{emitter-branch}
      \draw ({-1.5*\SCfactor},{-1.5*\SCfactor})--({3*\SCfactor-1},{-3.5*\SCfactor});
    \end{SCpart}
    \begin{SCpart}{emitter-arrow}
      \begin{scope}[shift={({-1.5*\SCfactor},{-1.5*\SCfactor})},rotate={\SCbranchangle+180}]
        \fill ({-1.05*\SCfactor},{0.55*\SCfactor})--({0*\SCfactor},{0*\SCfactor})--({-1.05*\SCfactor},{-0.55*\SCfactor})--cycle;
      \end{scope}
    \end{SCpart}
  \end{scope}
\begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm]
\begin{SCpart}{enclosure}
\draw ({0.5*#2},0) circle ({5.6*#2});
\end{SCpart}
\end{scope}
}
