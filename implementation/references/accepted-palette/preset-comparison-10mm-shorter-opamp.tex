% Extend each signal lead by 1 mm from the previous palette; supply leads are unchanged.
\newcommand{\sample}[4]{%
  \begin{scope}[shift={(#1,0)},line width=#3mm,line cap=round,line join=round]
    \pgfmathsetmacro{\s}{#2*5/3}
    \pgfmathsetmacro{\passiveEnd}{ceil(3*\s+2)+1}
    \pgfmathsetmacro{\capEnd}{ceil(0.6*\s+2)+1}
    \pgfmathsetmacro{\ampHalfWidth}{3.4*\s}
    \pgfmathsetmacro{\ampEnd}{ceil(\ampHalfWidth+2)+1}
    \pgfmathsetmacro{\supplyEnd}{ceil(2*\s+2)}
    \ifnum#4=0
      \draw ({-3*\s},-\s) rectangle ({3*\s},\s);
      \draw (-\passiveEnd,0)--({-3*\s},0) ({3*\s},0)--(\passiveEnd,0);
    \fi
    \ifnum#4=1
      \draw (-\passiveEnd,0)--({-3*\s},0)--({-2.5*\s},\s)--({-1.5*\s},-\s)--({-0.5*\s},\s)--({0.5*\s},-\s)--({1.5*\s},\s)--({2.5*\s},-\s)--({3*\s},0)--(\passiveEnd,0);
    \fi
    \ifnum#4=2
      \draw (-\passiveEnd,0)--({-3*\s},0);
      \draw ({-3*\s},0) arc[start angle=180,end angle=0,x radius={0.75*\s},y radius={1.2*\s}]
        arc[start angle=180,end angle=0,x radius={0.75*\s},y radius={1.2*\s}]
        arc[start angle=180,end angle=0,x radius={0.75*\s},y radius={1.2*\s}]
        arc[start angle=180,end angle=0,x radius={0.75*\s},y radius={1.2*\s}];
      \draw ({3*\s},0)--(\passiveEnd,0);
    \fi
    \ifnum#4=3
      \draw ({-0.6*\s},{-2*\s})--({-0.6*\s},{2*\s}) ({0.6*\s},{-2*\s})--({0.6*\s},{2*\s});
      \draw (-\capEnd,0)--({-0.6*\s},0) ({0.6*\s},0)--(\capEnd,0);
    \fi
    \ifnum#4>3
      % Triangle width reduced 15%; height and sign proportions retained.
      \draw (-\ampHalfWidth,{-4*\s})--(-\ampHalfWidth,{4*\s})--(\ampHalfWidth,0)--cycle;
      % Input row spacing remains 6 mm; lead ends conform to the 1 mm grid.
      \draw (-\ampEnd,3)--(-\ampHalfWidth,3) (-\ampEnd,-3)--(-\ampHalfWidth,-3) (\ampHalfWidth,0)--(\ampEnd,0);
      \pgfmathsetmacro{\markX}{-\ampHalfWidth+\s}
      \draw ({\markX-0.4*\s},3)--({\markX+0.4*\s},3) (\markX,{3-0.4*\s})--(\markX,{3+0.4*\s});
      \draw ({\markX-0.4*\s},-3)--({\markX+0.4*\s},-3);
      \ifnum#4=5
        \draw (0,{2*\s})--(0,\supplyEnd) (0,{-2*\s})--(0,-\supplyEnd);
        \fill[pinblue] (0,\supplyEnd) circle[radius=0.14] (0,-\supplyEnd) circle[radius=0.14];
      \fi
      \fill[pinblue] (-\ampEnd,3) circle[radius=0.14] (-\ampEnd,-3) circle[radius=0.14] (\ampEnd,0) circle[radius=0.14];
    \else
      \ifnum#4=3
        \fill[pinblue] (-\capEnd,0) circle[radius=0.14] (\capEnd,0) circle[radius=0.14];
      \else
        \fill[pinblue] (-\passiveEnd,0) circle[radius=0.14] (\passiveEnd,0) circle[radius=0.14];
      \fi
    \fi
  \end{scope}%
}
\newcommand{\samplerow}[5]{%
  \begin{scope}[shift={(0,#1)}]
    \node[anchor=east,align=right,font=\fontsize{8}{9}\selectfont] at (-15,0) {#4\\{\fontsize{6.5}{8}\selectfont #5}};
    \foreach \x/\kind in {0/0,28/1,56/2,84/3,112/4,140/5}{
      \begin{scope}[shift={(\x,0)}]
        \draw[step=1,black!15,line width=0.035mm] (-12,-10) grid (12,10);
        \draw[step=5,black!22,line width=0.045mm] (-12,-10) grid (12,10);
      \end{scope}
      \sample{\x}{#2}{#3}{\kind}
    }
  \end{scope}%
}
