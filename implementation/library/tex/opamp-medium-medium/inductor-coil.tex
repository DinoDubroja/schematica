% Accepted palette kind 2; four elliptical half-turns.
\newcommand{\SCInductorCoil}[3]{%
  \SCresolve{#1}{#2}{#3}%
  \pgfmathsetmacro{\SCend}{ceil(3*\SCscale+2)+1}%
  \begin{scope}[x=1mm,y=1mm,line width=\SCstroke mm,line cap=round,line join=round]
    \SCpin{a}{-\SCend}{0}\SCpin{b}{\SCend}{0}
    \SCdimension{body-width}{6*\SCscale}\SCdimension{body-height}{1.2*\SCscale}
    \begin{SCpart}{lead-a}\draw (-\SCend,0)--({-3*\SCscale},0);\end{SCpart}
    \begin{SCpart}{body}
      \draw ({-3*\SCscale},0) arc[start angle=180,end angle=0,x radius={0.75*\SCscale},y radius={1.2*\SCscale}]
        arc[start angle=180,end angle=0,x radius={0.75*\SCscale},y radius={1.2*\SCscale}]
        arc[start angle=180,end angle=0,x radius={0.75*\SCscale},y radius={1.2*\SCscale}]
        arc[start angle=180,end angle=0,x radius={0.75*\SCscale},y radius={1.2*\SCscale}];
    \end{SCpart}
    \begin{SCpart}{lead-b}\draw ({3*\SCscale},0)--(\SCend,0);\end{SCpart}
  \end{scope}%
}
